### Task 2: Crowd on the web (export + instanced render)

**Files:**
- Create: `scripts/webexport/crowdbin.py`, `tests/webexport/test_crowdbin.py`, `web/src/crowd.ts`, `web/src/crowd.test.ts`

**Interfaces:**
- Produces, file format `crowd.bin` (little-endian):
  - header `<4sII`: magic `b"CRWD"`, version `1`, count;
  - then `count` records `<fffBB`: x, y, z (Blender metres, Z up), `look` (0 White, 1 Dark, 2 Green, 3 Mixed, 4 Flag) and `yaw` (0–255 → 0–2π).
  - Record size 14 bytes.
- Produces, manifest: `"crowd": "crowd.bin"` (optional key).
- Produces, Python: `crowdbin.pack(rows) -> bytes`, `crowdbin.unpack(blob) -> list[tuple[float, float, float, int, float]]`, `LOOKS = ("White", "Dark", "Green", "Mixed", "Flag")`.
- Produces, TS:
  - `parseCrowd(buf: ArrayBuffer | null): CrowdData | null`, where `CrowdData = { count: number; pos: Float32Array; look: Uint8Array; yaw: Float32Array }`;
  - `buildCrowd(d: CrowdData, u: TreeUniforms): THREE.Group`.

`scripts/webexport/crowdbin.py`:

```python
"""crowd.bin: the fans placed by stadium/site.py (Crowd_Person_* point meshes), for the web viewer.

header  <4sII   magic "CRWD", version, count
record  <fffBB  x, y, z (Blender metres, Z up), look (index into LOOKS), yaw (0..255 -> 0..2 pi)
"""
import math
import struct

MAGIC, VERSION = b"CRWD", 1
HEADER_FMT, RECORD_FMT = "<4sII", "<fffBB"
HEADER, RECORD = struct.calcsize(HEADER_FMT), struct.calcsize(RECORD_FMT)
assert (HEADER, RECORD) == (12, 14)
LOOKS = ("White", "Dark", "Green", "Mixed", "Flag")


def pack(rows):
    out = [struct.pack(HEADER_FMT, MAGIC, VERSION, len(rows))]
    for x, y, z, look, yaw in rows:
        if not 0 <= look < len(LOOKS):
            raise ValueError(f"look {look} out of range")
        q = int(round((yaw % (2 * math.pi)) / (2 * math.pi) * 255)) % 256
        out.append(struct.pack(RECORD_FMT, x, y, z, look, q))
    return b"".join(out)


def unpack(blob):
    magic, version, n = struct.unpack_from(HEADER_FMT, blob, 0)
    assert magic == MAGIC and version == VERSION
    return [(x, y, z, look, q / 255 * 2 * math.pi)
            for x, y, z, look, q in struct.iter_unpack(RECORD_FMT, blob[HEADER:HEADER + n * RECORD])]
```

def stage_crowd(report, quality=None):
    """The fans (site.py crowds) as crowd.bin; the web draws them instanced (plan 2026-09-28 Task 2)."""
    import random
    from webexport import crowdbin, manifest
    rows, rng = [], random.Random(3)
    for look, name in enumerate(crowdbin.LOOKS):
        o = bpy.data.objects.get(f"Crowd_Person_{name}")
        if o is None:
            continue
        for v in o.data.vertices:
            x, y, z = o.matrix_world @ v.co
            rows.append((x, y, z, look, rng.uniform(0, 6.2832)))
    with open(os.path.join(RAW, "crowd.bin"), "wb") as f:
        f.write(crowdbin.pack(rows))
    manifest.merge(os.path.join(RAW, "manifest.json"), {"crowd": "crowd.bin"})
    report["crowd"] = len(rows)

`web/src/crowd.ts`:

```ts
import * as THREE from "three";
import { toThree } from "./coords";
import { patchOutdoorLight, type TreeUniforms } from "./treeLight";

/** Look index -> body colour (linear), as landscape.crowds; 4 = fan with a green flag. */
export const CROWD_LOOKS = [[0.85, 0.84, 0.8], [0.03, 0.03, 0.035], [0.02, 0.35, 0.16], [0.35, 0.3, 0.26], [0.85, 0.84, 0.8]] as const;
export const FLAG_COLOR = [0.0, 0.33, 0.14] as const;
export interface CrowdData { count: number; pos: Float32Array; look: Uint8Array; yaw: Float32Array }

export function parseCrowd(buf: ArrayBuffer | null): CrowdData | null {
  if (!buf || buf.byteLength < 12) return null;
  const v = new DataView(buf);
  if (String.fromCharCode(v.getUint8(0), v.getUint8(1), v.getUint8(2), v.getUint8(3)) !== "CRWD" || v.getUint32(4, true) !== 1) return null;
  const n = v.getUint32(8, true), pos = new Float32Array(3 * n), look = new Uint8Array(n), yaw = new Float32Array(n);
  for (let i = 0; i < n; i++) {
    const o = 12 + 14 * i;
    pos[3 * i] = v.getFloat32(o, true); pos[3 * i + 1] = v.getFloat32(o + 4, true); pos[3 * i + 2] = v.getFloat32(o + 8, true);
    look[i] = v.getUint8(o + 12); yaw[i] = (v.getUint8(o + 13) / 255) * 2 * Math.PI;
  }
  return { count: n, pos, look, yaw };
}

/** Body (6-sided cone, 1.45 m) + head, Y-up metres; the flag look adds a pole and a flag. */
function figure(flag: boolean): THREE.BufferGeometry {
  const parts: THREE.BufferGeometry[] = [
    new THREE.CylinderGeometry(0.2, 0.26, 1.45, 6).translate(0, 0.72, 0),
    new THREE.OctahedronGeometry(0.13).translate(0, 1.58, 0),
  ];
  if (flag) {
    parts.push(new THREE.CylinderGeometry(0.02, 0.02, 1.3, 3).translate(0.3, 1.75, 0));
    parts.push(new THREE.PlaneGeometry(0.9, 0.55).translate(0.75, 2.1, 0.04));
  }
  const g = mergeGeometries(parts);
  if (flag) {                                            // group 1 = the flag cloth, for its own material
    const flagTris = 2, total = g.index ? g.index.count : g.attributes.position.count;
    g.clearGroups(); g.addGroup(0, total - flagTris * 3, 0); g.addGroup(total - flagTris * 3, flagTris * 3, 1);
  }
  return g;
}
```

`mergeGeometries` comes from `three/examples/jsm/utils/BufferGeometryUtils.js`; import it at the top of the file. Every part must be non-indexed, or every part indexed, before merging. Call `.toNonIndexed()` on each part if the merge refuses.

```ts
export function buildCrowd(d: CrowdData, u: TreeUniforms): THREE.Group {
  const g = new THREE.Group(); g.name = "crowd";
  const m4 = new THREE.Matrix4(), q = new THREE.Quaternion(), s = new THREE.Vector3(1, 1, 1), up = new THREE.Vector3(0, 1, 0);
  for (let look = 0; look < CROWD_LOOKS.length; look++) {
    const idx: number[] = []; for (let i = 0; i < d.count; i++) if (d.look[i] === look) idx.push(i);
    if (!idx.length) continue;
    const body = new THREE.MeshStandardMaterial({ color: new THREE.Color(...CROWD_LOOKS[look]), roughness: 0.8 });
    patchOutdoorLight(body, u, "crowd");
    const mats: THREE.Material[] = [body];
    if (look === 4) { const f = new THREE.MeshStandardMaterial({ color: new THREE.Color(...FLAG_COLOR), roughness: 0.7, side: THREE.DoubleSide }); patchOutdoorLight(f, u, "crowdFlag"); mats.push(f); }
    const mesh = new THREE.InstancedMesh(figure(look === 4), mats.length > 1 ? mats : body, idx.length);
    idx.forEach((i, k) => {
      const p = toThree([d.pos[3 * i], d.pos[3 * i + 1], d.pos[3 * i + 2]]);
      q.setFromAxisAngle(up, d.yaw[i]);
      mesh.setMatrixAt(k, m4.compose(p, q, s));
    });
    mesh.instanceMatrix.needsUpdate = true;
    mesh.raycast = () => undefined;                   // never steals seat / section picks (Review Focus 5)
    mesh.castShadow = false; mesh.receiveShadow = true;
    g.add(mesh);
  }
  return g;
}
```
